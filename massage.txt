local function GetLuckBarPercent()
    local ui = LocalPlayer:FindFirstChild("PlayerGui")
    if not ui then return 0 end
    local folder = ui:FindFirstChild("UiFolder")
    if not folder then return 0 end
    local main = folder:FindFirstChild("Main")
    if not main then return 0 end
    local hud = main:FindFirstChild("HUD")
    if not hud then return 0 end
    local luckFrame = hud:FindFirstChild("Luck") or hud:FindFirstChild("LuckBar")
    if not luckFrame then return 0 end
    local bar = luckFrame:FindFirstChild("Bar") or luckFrame:FindFirstChild("Fill")
    if bar and bar:IsA("Frame") then
        return math.floor(bar.Size.X.Scale * 100)
    end
    local textLabel = luckFrame:FindFirstChild("Text") or luckFrame:FindFirstChild("Percent")
    if textLabel and textLabel:IsA("TextLabel") then
        local num = tonumber(textLabel.Text:gsub("[^%d]", ""))
        return num or 0
    end
    return 0
end

local function GetLandingPos()
    local target = GetCoinTarget()
    if not target then return nil end
    local char = LocalPlayer.Character
    if not char then return nil end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local pPos = hrp.Position
    local fPos = target.Position
    local dir = Vector3.new(pPos.X, fPos.Y, pPos.Z) - fPos
    if dir.Magnitude < 0.05 then dir = Vector3.new(0, 0, 1) else dir = dir.Unit end
    return fPos + dir * 8
end

--// FIXED: Get Item Rarity (Throw a Coin specific)
local function GetItemRarity(itemName)
    if not itemName then return "Common" end
    local name = itemName:lower()

    -- Check for mutation suffixes first
    if name:find(": divine") or name:find("divine") then
        return "Divine"
    end
    if name:find(": astral") or name:find("astral") then
        return "Celestial"
    end
    if name:find(": rainbow") or name:find("rainbow") then
        return "Mythic"
    end
    if name:find(": huge") or name:find("huge") then
        return "Legendary"
    end
    if name:find(": big") or name:find("big") then
        return "Epic"
    end

    -- Base rarity detection
    if name:find("celestial") then return "Celestial" end
    if name:find("divine") then return "Divine" end
    if name:find("mythic") then return "Mythic" end
    if name:find("legendary") then return "Legendary" end
    if name:find("epic") then return "Epic" end
    if name:find("rare") then return "Rare" end
    if name:find("uncommon") then return "Uncommon" end
    return "Common"
end

--// ENSURE IN RANGE
local function EnsureInRange()
    local target = GetCoinTarget()
    if not target then return false end
    local char = LocalPlayer.Character
    if not char then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    local dist = (hrp.Position - target.Position).Magnitude
    if dist <= 50 then return true end
    if dist > 50 and getgenv().TAC_Config.AutoApproach and not approaching then
        approaching = true
        local dir = (hrp.Position - target.Position)
        if dir.Magnitude < 0.01 then dir = Vector3.new(0,0,1) else dir = dir.Unit end
        local targetPos = target.Position + dir * 30
        targetPos = Vector3.new(targetPos.X, math.max(targetPos.Y, target.Position.Y + 5), targetPos.Z)
        hrp.CFrame = CFrame.new(targetPos, target.Position)
        task.wait(0.2)
        approaching = false
        return true
    end
    return dist <= getgenv().TAC_Config.MaxThrowDistance
end

--// NORMAL THROW
local function DoThrow()
    local cfg = getgenv().TAC_Config
    if not cfg.AutoThrow then return end
    if throwCooldown then return end
    if throwInProgress then return end
    if approaching then return end
    if cfg.SpamMode then return end

    local now = tick()
    if now - lastThrow < 3.0 then return end

    local coin = GetEquippedCoin()
    if not coin then
        local claimStarter = Events:FindFirstChild("ClaimStarterCash")
        if claimStarter then claimStarter:FireServer() end
        return
    end

    if not EnsureInRange() then return end
    local pos = GetLandingPos()
    if not pos then return end

    throwInProgress = true
    lastThrow = now
    throwId = throwId + 1
    sessionThrows = sessionThrows + 1

    local isDouble = LocalPlayer:GetAttribute("DoubleThrow") == true
    local pos2 = nil
    local throwPos = pos

    if isDouble then
        local target = GetCoinTarget()
        if not target then throwInProgress = false return end
        local flatDir = (pos - target.Position) * Vector3.new(1,0,1)
        if flatDir.Magnitude > 0 then flatDir = flatDir.Unit else flatDir = Vector3.new(0,0,1) end
        local perp = Vector3.new(-flatDir.Z, 0, flatDir.X)
        throwPos = pos + perp * 3
        pos2 = pos - perp * 3
    end

    pcall(function()
        if isDouble then
            CoinThrow:FireServer(coin, throwPos, pos2)
        else
            CoinThrow:FireServer(coin, throwPos)
        end
    end)

    task.delay(1.2, function()
        if not cfg.AutoThrow then throwInProgress = false return end
        pcall(function()
            local desiredLuck = tonumber(LocalPlayer:GetAttribute("DesiredLuck")) or 0
            local finalId = (serverThrowId > 0 and serverThrowId) or throwId
            CoinLanded:FireServer(1, pos, coin, pos2, desiredLuck, finalId)
        end)
        throwInProgress = false
    end)

    task.delay(5, function() throwInProgress = false end)
end

--// SPAM THROW
local function StartSpamThrow()
    local cfg = getgenv().TAC_Config
    if spamThread then return end
    spamThread = task.spawn(function()
        while cfg.AutoThrow and cfg.SpamMode do
            local coin = GetEquippedCoin()
            if coin then
                local pos = GetLandingPos()
                if pos then
                    local isDouble = LocalPlayer:GetAttribute("DoubleThrow") == true
                    local pos2 = nil
                    local throwPos = pos
                    if isDouble then
                        local target = GetCoinTarget()
                        if target then
                            local flatDir = (pos - target.Position) * Vector3.new(1,0,1)
                            if flatDir.Magnitude > 0 then flatDir = flatDir.Unit else flatDir = Vector3.new(0,0,1) end
                            local perp = Vector3.new(-flatDir.Z, 0, flatDir.X)
                            throwPos = pos + perp * 3
                            pos2 = pos - perp * 3
                        end
                    end
                    pcall(function()
                        if isDouble then
                            CoinThrow:FireServer(coin, throwPos, pos2)
                        else
                            CoinThrow:FireServer(coin, throwPos)
                        end
                    end)
                    task.wait(0.3)
                    pcall(function()
                        local desiredLuck = tonumber(LocalPlayer:GetAttribute("DesiredLuck")) or 0
                        throwId = throwId + 1
                        if isDouble then
                            CoinLanded:FireServer(3, throwPos, coin, pos2, desiredLuck, throwId)
                        else
                            CoinLanded:FireServer(3, throwPos, coin, nil, desiredLuck, throwId)
                        end
                        sessionThrows = sessionThrows + 1
                    end)
                end
            end
            task.wait(cfg.SpamInterval)
        end
        spamThread = nil
    end)
end

local function StopSpamThrow()
    spamThread = nil
end

--// AUTO BUY
local function DoAutoBuy()
    local cfg = getgenv().TAC_Config
    if not cfg.AutoBuy then return end
    local now = tick()
    if now - lastBuy < cfg.BuyDelay then return end
    local cash = GetCash()
    local target = nil

    if cfg.BuyMode == "Best" then
        local bestLuck = -1
        for name, data in pairs(CoinsData) do
            if not OwnedCoins[name] then
                local cost = data.Cost or 0
                local world = data.World or 1
                if cash >= cost and CurrentWorld >= world then
                    if data.Luck > bestLuck then bestLuck = data.Luck target = name end
                end
            end
        end
    else
        target = cfg.TargetCoin
        if target and CoinsData[target] then
            if OwnedCoins[target] then return end
            local cost = CoinsData[target].Cost or 0
            local world = CoinsData[target].World or 1
            if cash < cost or CurrentWorld < world then return end
        else
            return
        end
    end

    if target then lastBuy = now BuyCoin:FireServer(target) end
end

--// AUTO EQUIP
local function DoAutoEquip()
    local cfg = getgenv().TAC_Config
    if not cfg.AutoEquipBest then return end
    local bestLuck = -1
    local bestCoin = nil
    for name, _ in pairs(OwnedCoins) do
        local data = CoinsData[name]
        if data and data.Luck > bestLuck then bestLuck = data.Luck bestCoin = name end
    end
    if bestCoin and GetEquippedCoin() ~= bestCoin then 
        pcall(function()
            EquipCoin:FireServer(bestCoin)
        end)
    end
end

local function DoAutoPotion()
    local cfg = getgenv().TAC_Config
    if not cfg.AutoPotion then return end
    local now = tick()
    if now - lastPotionUse < POTION_INTERVAL then return end

    if UsePotion then
        local potionName = cfg.PotionType
        if potionName == "10xLuck" then potionName = "10x Luck Potion" end
        if potionName == "5xLuck" then potionName = "5x Luck Potion" end
        if potionName == "Luck" then potionName = "Luck Potion" end

        pcall(function()
            UsePotion:FireServer(potionName)
            lastPotionUse = now
            notify("Potion", "Used: "..potionName)
        end)
    else
        -- Fallback: Try to use potion from UI
        local ui = LocalPlayer:FindFirstChild("PlayerGui")
        if not ui then return end
        for _, gui in pairs(ui:GetChildren()) do
            if gui:IsA("ScreenGui") then
                for _, obj in pairs(gui:GetDescendants()) do
                    if obj:IsA("TextButton") and obj.Visible then
                        local text = obj.Text
                        if text and (text:lower():find("luck") or text:lower():find("potion")) and text:lower():find("use") then
                            pcall(function()
                                obj.MouseButton1Click:Fire()
                                lastPotionUse = now
                            end)
                            return
                        end
                    end
                end
            end
        end
    end
end

--// ═══════════════════════════════════════════════════════════════
local function GetInventoryItems()
    local items = {}
    local ui = LocalPlayer:FindFirstChild("PlayerGui")
    if not ui then return items end
    for _, gui in pairs(ui:GetChildren()) do
        if gui:IsA("ScreenGui") and (gui.Name:lower():find("inv") or gui.Name:lower():find("backpack")) then
            for _, frame in pairs(gui:GetDescendants()) do
                if frame:IsA("TextLabel") and (frame.Name:lower():find("name") or frame.Name:lower():find("item")) then
                    if frame.Text and frame.Text ~= "" then
                        table.insert(items, frame.Text)
                    end
                end
            end
        end
    end
    return items
end

local function CountItemDuplicates(items)
    local counts = {}
    for _, item in pairs(items) do
        counts[item] = (counts[item] or 0) + 1
    end
    return counts
end

local function DoAutoMutate()
    local cfg = getgenv().TAC_Config
    if not cfg.AutoMutate then return end
    local now = tick()
    if now - lastMutate < MUTATE_INTERVAL then return end

    if MutateItem then
        local items = GetInventoryItems()
        local counts = CountItemDuplicates(items)
        for itemName, count in pairs(counts) do
            if count >= cfg.MutateMinCopies then
                local rarity = GetItemRarity(itemName)
                if (RARITY_ORDER[rarity] or 1) >= (RARITY_ORDER[cfg.MutateMinRarity] or 1) then
                    pcall(function()
                        MutateItem:FireServer(itemName, cfg.MutateMinCopies)
                        lastMutate = now
                        notify("Mutation", "Mutating "..itemName.." x"..cfg.MutateMinCopies)
                    end)
                    return
                end
            end
        end
        return
    end

    -- Method 2: Find Mutation Machine in workspace
    local machine = nil
    for _, obj in pairs(Workspace:GetDescendants()) do
        if obj.Name:lower():find("mutation") or obj.Name:lower():find("mutate") then
            machine = obj
            break
        end
    end

    if machine then
        local prompt = machine:FindFirstChildOfClass("ProximityPrompt")
        if prompt then
            pcall(function()
                fireproximityprompt(prompt)
                lastMutate = now
            end)
        end
        local clickDetector = machine:FindFirstChildOfClass("ClickDetector")
        if clickDetector then
            pcall(function()
                fireclickdetector(clickDetector)
                lastMutate = now
            end)
        end
    end
end

--// AUTO REDEEM CODES
--// ═══════════════════════════════════════════════════════════════
local function DoAutoRedeemCodes()
    local cfg = getgenv().TAC_Config
    if not cfg.AutoRedeemCodes then return end
    local now = tick()
    if now - lastRedeem < REDEEM_INTERVAL then return end

    if RedeemCode then
        for _, code in pairs(WORKING_CODES) do
            if not redeemedCodes[code] then
                pcall(function()
                    RedeemCode:FireServer(code)
                    redeemedCodes[code] = true
                    notify("Code Redeemed", code)
                end)
                task.wait(1)
            end
        end
        lastRedeem = now
        return
    end

    -- UI Automation fallback
    local ui = LocalPlayer:FindFirstChild("PlayerGui")
    if not ui then return end

    for _, gui in pairs(ui:GetChildren()) do
        if gui:IsA("ScreenGui") then
            for _, obj in pairs(gui:GetDescendants()) do
                if obj:IsA("TextBox") and (obj.Name:lower():find("code") or (obj.PlaceholderText and obj.PlaceholderText:lower():find("code"))) then
                    for _, code in pairs(WORKING_CODES) do
                        if not redeemedCodes[code] then
                            pcall(function()
                                obj.Text = code
                                task.wait(0.3)
                                for _, btn in pairs(gui:GetDescendants()) do
                                    if btn:IsA("TextButton") and (btn.Text:lower():find("claim") or btn.Text:lower():find("redeem")) then
                                        pcall(function() btn.MouseButton1Click:Fire() end)
                                        redeemedCodes[code] = true
                                        notify("Code", code.." redeemed!")
                                        break
                                    end
                                end
                            end)
                            task.wait(1)
                        end
                    end
                    lastRedeem = now
                    return
                end
            end
        end
    end

    lastRedeem = now
end

            Content = "Last Sell: " .. lastSellText
        })

        local cash = GetCash()

            Content = "Cash: $" .. cash .. " | Owned: " .. ownedCount
        })

        local luckStatus = cfg.AutoUpgradeLuck and "ON" or "OFF"
        local valueStatus = cfg.AutoUpgradeValue and "ON" or "OFF"
        local speedStatus = cfg.AutoUpgradeSpeed and "ON" or "OFF"

            Content = "Auto Upgrade Status"
        })

        local potionStatus = cfg.AutoPotion and "ON" or "OFF"
